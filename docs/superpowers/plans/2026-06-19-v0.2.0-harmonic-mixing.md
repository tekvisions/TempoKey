```swift
    /// Camelot wheel code, e.g. "8A" (minor) / "8B" (major).
    public var camelot: String {
        // number per pitch class (C…B); minor = A ring, major = B ring
        let major = [8, 3, 10, 5, 12, 7, 2, 9, 4, 11, 6, 1]
        let minor = [5, 12, 7, 2, 9, 4, 11, 6, 1, 8, 3, 10]
        let n = (mode == .major ? major : minor)[tonic.rawValue]
        return "\(n)\(mode == .major ? "B" : "A")"
    }

    /// The three harmonically-compatible Camelot codes: ±1 on the same ring, plus the relative
    /// (same number, other ring).
    public var compatibleCamelot: [String] {
        let letter = mode == .major ? "B" : "A"
        let other = mode == .major ? "A" : "B"
        let n = Int(camelot.dropLast()) ?? 1
        let down = (n + 10) % 12 + 1   // n-1 with 1→12 wrap
        let up = n % 12 + 1            // n+1 with 12→1 wrap
        return ["\(down)\(letter)", "\(up)\(letter)", "\(n)\(other)"]
    }
```

```swift
import Foundation

/// Averages the intervals between recent taps into a BPM. A gap > 2 s starts a new sequence.
struct TapTempo {
    private var taps: [TimeInterval] = []

    mutating func tap(at time: TimeInterval) -> Double? {
        if let last = taps.last, time - last > 2.0 { taps.removeAll() }
        taps.append(time)
        guard taps.count >= 2 else { return nil }
        let intervals = zip(taps.dropFirst(), taps).map { $0 - $1 }
        let avg = intervals.reduce(0, +) / Double(intervals.count)
        return avg > 0 ? 60.0 / avg : nil
    }

    mutating func reset() { taps.removeAll() }
}
```

```swift
import Foundation
import TempoKeyCore

/// User corrections applied to the *displayed* tempo/key only — the analysis result is untouched.
struct DisplayOverrides: Equatable {
    var bpmMultiplier: Double = 1     // 0.5, 1, or 2
    var manualBPM: Double?            // from tap-tempo
    var manualKey: MusicalKey?

    func displayedBPM(detected: Double) -> Double { (manualBPM ?? detected) * bpmMultiplier }
    func displayedKey(detected: MusicalKey) -> MusicalKey { manualKey ?? detected }
    var isActive: Bool { bpmMultiplier != 1 || manualBPM != nil || manualKey != nil }
    mutating func clear() { self = DisplayOverrides() }
}
```
